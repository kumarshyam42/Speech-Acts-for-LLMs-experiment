```python
def to_csv(expenses):
    buffer = io.StringIO()
    writer = csv.DictWriter(buffer, fieldnames=CSV_FIELDS)
    writer.writeheader()
    for exp in expenses:
        row = {field: exp.get(field, "") for field in CSV_FIELDS}
        writer.writerow(row)
    return buffer.getvalue()
```
